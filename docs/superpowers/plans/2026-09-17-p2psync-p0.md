```ts
/**
 * 房间 ID 与分享链接工具。
 * 房间 ID：10 字节随机数编码为 16 字符 Base32（约 80bit 熵），即房间准入凭证。
 * 分享链接：p2psync://join?room=<roomId>&url=<encodeURIComponent(视频页地址)>
 */

const BASE32_ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567'
const ROOM_ID_PATTERN = /^[A-Z2-7]{16}$/

/**
 * 生成房间 ID。
 * 返回值：16 字符 Base32 字符串（如 "K7Q2M9XT4P1WVBEH"）。
 */
export function generateRoomId(): string {
  const bytes = new Uint8Array(10)
  crypto.getRandomValues(bytes)
  let bits = 0
  let val = 0
  let out = ''
  for (const b of bytes) {
    val = (val << 8) | b
    bits += 8
    while (bits >= 5) {
      out += BASE32_ALPHABET[(val >>> (bits - 5)) & 31]
      bits -= 5
    }
  }
  return out
}

/**
 * 生成分享链接。
 * 参数：roomId 房间 ID；videoUrl 视频页地址（http/https）。
 * 返回值：p2psync:// 协议链接。
 */
export function buildShareUrl(roomId: string, videoUrl: string): string {
  return `p2psync://join?room=${roomId}&url=${encodeURIComponent(videoUrl)}`
}

/**
 * 解析分享链接。
 * 参数：input 用户粘贴或系统传来的链接。
 * 返回值：解析成功返回 { roomId, videoUrl }；协议错误、房间号非法、视频地址非 http(s) 时返回 null。
 */
export function parseShareUrl(input: string): { roomId: string; videoUrl: string } | null {
  try {
    const u = new URL(input)
    if (u.protocol !== 'p2psync:' || u.host !== 'join') return null
    const roomId = u.searchParams.get('room') ?? ''
    const videoUrl = u.searchParams.get('url') ?? ''
    if (!ROOM_ID_PATTERN.test(roomId)) return null
    const v = new URL(videoUrl)
    if (v.protocol !== 'http:' && v.protocol !== 'https:') return null
    return { roomId, videoUrl }
  } catch {
    return null
  }
}
```

`core/protocol.ts`:

```ts
/**
 * 同步消息协议（DataChannel 应用层）。
 * at: 发送方墙钟时间戳（ms），接收方据此推算当前应处的播放位置。
 */
export type SyncMsg =
  /** 新成员请求全量状态（成员→房主） */
  | { t: 'hello' }
  /** 全量状态/周期心跳（房主→全员） */
  | { t: 'state'; url: string; position: number; playing: boolean; at: number }
  /** 房主触发播放 */
  | { t: 'play'; position: number; at: number }
  /** 房主触发暂停 */
  | { t: 'pause'; position: number }
  /** 房主拖动进度 */
  | { t: 'seek'; position: number; playing: boolean; at: number }

/** 消息类型到必检数值字段的映射，用于解码校验 */
const NUMERIC_FIELDS: Record<SyncMsg['t'], string[]> = {
  hello: [],
  state: ['position', 'at'],
  play: ['position', 'at'],
  pause: ['position'],
  seek: ['position', 'at'],
}

/**
 * 编码消息为字符串（DataChannel 载荷）。
 * 参数：msg 协议消息。
 * 返回值：JSON 字符串。
 */
export function encodeMsg(msg: SyncMsg): string {
  return JSON.stringify(msg)
}

/**
 * 解码并校验消息。
 * 参数：raw 对端发来的字符串。
 * 返回值：合法返回 SyncMsg，任何字段缺失/类型错误/未知类型返回 null。
 */
export function decodeMsg(raw: string): SyncMsg | null {
  try {
    const o = JSON.parse(raw) as Record<string, unknown>
    if (typeof o !== 'object' || o === null || !(o.t in NUMERIC_FIELDS)) return null
    const t = o.t as SyncMsg['t']
    for (const f of NUMERIC_FIELDS[t]) {
      const v = o[f]
      if (typeof v !== 'number' || !Number.isFinite(v)) return null
    }
    if (t === 'state' && typeof o.url !== 'string') return null
    if ((t === 'state' || t === 'seek') && typeof o.playing !== 'boolean') return null
    return o as unknown as SyncMsg
  } catch {
    return null
  }
}
```

`core/syncEngine.ts`:

```ts
/**
 * 同步引擎纯逻辑（成员侧）：
 * 根据房主心跳推算理论位置，与本地实际位置比较后给出动作决策。
 * 阈值：偏差 > SEEK_THRESHOLD 秒直接 seek；否则用 playbackRate 在 0.95~1.05 内微调。
 */

/** 直接跳转阈值（秒） */
const SEEK_THRESHOLD = 0.35
/** 速率微调生效阈值（秒），小于它不动作 */
const RATE_THRESHOLD = 0.15
/** 速率微调时间窗（秒）：偏差/窗口 = 目标速率 */
const RATE_WINDOW = 4
/** 速率限幅 */
const RATE_MIN = 0.95
const RATE_MAX = 1.05

/** 房主心跳/状态快照 */
export interface StateSnapshot {
  /** 心跳时刻的播放位置（秒） */
  position: number
  /** 心跳时刻是否播放中 */
  playing: boolean
  /** 房主墙钟时间戳（ms） */
  at: number
}

/** 校正决策 */
export type Correction =
  | { kind: 'none' }
  | { kind: 'rate'; rate: number }
  | { kind: 'seek'; position: number }

/**
 * 推算当前应处的播放位置。
 * 参数：s 房主快照；now 本机当前时间（ms）。
 * 返回值：理论播放位置（秒）；暂停时等于快照位置。
 */
export function computeTargetPosition(s: StateSnapshot, now: number): number {
  return s.position + (s.playing ? (now - s.at) / 1000 : 0)
}

/**
 * 依据理论位置与实际位置给出校正决策。
 * 参数：target 理论位置；actual 本地实际位置（秒）。
 * 返回值：Correction 决策。
 */
export function decideCorrection(target: number, actual: number): Correction {
  const diff = target - actual // 正值=本地落后，需加速/快进
  if (Math.abs(diff) > SEEK_THRESHOLD) return { kind: 'seek', position: target }
  if (Math.abs(diff) <= RATE_THRESHOLD) return { kind: 'none' }
  const rate = 1 + diff / RATE_WINDOW
  return { kind: 'rate', rate: Math.min(RATE_MAX, Math.max(RATE_MIN, rate)) }
}
```

```ts
 * 设计约束：其余模块不直接接触 Trystero API，未来更换信令策略只改本文件。
 * Trystero 房间对象通过参数注入，便于测试时替换为 fake。
 */
import { decodeMsg, encodeMsg, type SyncMsg } from './protocol'

/** Trystero 房间对象的最小接口（与 trystero/nostr joinRoom 返回值兼容） */
export interface TrysteroRoomLike {
  makeAction: (ns: string) => [send: (data: unknown, toPeer?: string) => void, receive: (cb: (data: unknown, peerId: string) => void) => void]
  onPeerJoin: (cb: (peerId: string) => void) => void
  onPeerLeave: (cb: (peerId: string) => void) => void
  leave: () => void
}

/** 应用命名空间：编入版本号，协议不兼容时切换，避免新旧串台 */
export const APP_ID = 'p2psync-v1'

/**
 * 创建房间句柄。
 * 参数：trysteroRoom 已 joinRoom 的房间对象；selfPeerId 本端 ID（仅用于日志）；onMessage 收到合法同步消息的回调。
 * 返回值：房间句柄（广播、事件订阅、leave 与原始 send）。
 */
export function createRoom(trysteroRoom: TrysteroRoomLike, selfPeerId: string, onMessage: (msg: SyncMsg, peerId: string) => void) {
  // 'sync' 命名空间承载全部 SyncMsg；Trystero 的 send 支持 JSON 可序列化数据，
  // 此处统一走字符串编码，解码校验集中在 protocol.ts
  const [rawSend, rawReceive] = trysteroRoom.makeAction('sync')
  trysteroRoom.onPeerJoin((id) => peers.add(id))
  trysteroRoom.onPeerLeave((id) => peers.delete(id))
  rawReceive((data, peerId) => {
    const msg = decodeMsg(String(data))
    if (msg) onMessage(msg, peerId)
  })
  const peers = new Set<string>()
  return {
    /** 向全员广播一条同步消息 */
    broadcast: (msg: SyncMsg) => rawSend(encodeMsg(msg)),
    /** 原始发送（保留给单播扩展） */
    rawSend,
    /** 房间内成员 ID 集合（不含本端） */
    peers,
    /** 订阅成员加入 */
    onPeerJoin: (cb: (id: string) => void) => trysteroRoom.onPeerJoin(cb),
    /** 订阅成员离开 */
    onPeerLeave: (cb: (id: string) => void) => trysteroRoom.onPeerLeave(cb),
    /** 离开房间并释放连接 */
    leave: () => trysteroRoom.leave(),
  }
}

/**
 * 打开真实房间（唯一与 trystero/nostr 耦合的位置）。
 * 参数：roomId 房间 ID。
 * 返回值：Trystero 房间对象。
 */
export async function openRealRoom(roomId: string): Promise<TrysteroRoomLike> {
  const { joinRoom } = await import('trystero/nostr')
  return joinRoom({ appId: APP_ID }, roomId) as unknown as TrysteroRoomLike
}
```

```ts
/**
 * 视频页注入脚本（以源码字符串导出，由主进程 webContents.executeJavaScript 执行）。
 * 桥对象 window.__p2pBridge：
 * - 事件方向（房主 monitor）：视频 play/pause/seeked 事件写入内部队列，主进程轮询 drain()
 * - 指令方向（成员 follower）：主进程调用 cmd(action) 执行 play/pause/seek/setRate
 * - guard(true)：成员端开启跟随模式，拦截网站对 video 的自动暂停/播放干扰（尽力而为）
 */

/** 房主端监听脚本：上报视频事件 */
export const MONITOR_SCRIPT = `
(() => {
  if (window.__p2pBridge) return 'already'
  const q = []
  const video = document.querySelector('video')
  if (!video) return 'novideo'
  const push = (ev, extra) => q.push({ ev, position: video.currentTime, paused: video.paused, ...extra })
  video.addEventListener('play', () => push('play'))
  video.addEventListener('pause', () => push('pause'))
  video.addEventListener('seeked', () => push('seek'))
  window.__p2pBridge = {
    drain: () => { const out = q.splice(0, q.length); return out },
    // 成员端指令执行：action = play|pause|seek|rate
    cmd: (action, arg) => {
      if (action === 'play') video.play().catch(() => {})
      if (action === 'pause') video.pause()
      if (action === 'seek') video.currentTime = arg
      if (action === 'rate') video.playbackRate = arg
    },
    // 跟随模式：捕获阶段吞掉网站自身的播放控制，避免与同步指令互搏
    guard: (on) => {
      if (on) video.addEventListener('play', (e) => { if (video.__p2pFollow) e.stopImmediatePropagation() }, true)
    },
    /** 查询当前视频状态（供周期校准与 UI 显示） */
    status: () => ({ position: video.currentTime, paused: video.paused, rate: video.playbackRate, duration: video.duration }),
    /** 标记"正在由同步指令驱动播放"，供 guard 区分来源 */
    setFollow: (v) => { video.__p2pFollow = v },
  }
  return 'ok'
})()
`

/** 成员端跟随模式脚本（在 MONITOR_SCRIPT 之后注入） */
export const FOLLOWER_GUARD_SCRIPT = `
(() => {
  const b = window.__p2pBridge
  if (!b) return 'nobridge'
  b.setFollow(true)
  b.guard(true)
  return 'ok'
})()
`
```

```ts
import { WebContentsView, type BrowserWindow } from 'electron'
import { FOLLOWER_GUARD_SCRIPT, MONITOR_SCRIPT } from '../inject/scripts'

/**
 * 视频页视图管理：在主窗口内加载视频网页，提供注入/轮询/指令通道。
 * 事件采用轮询 drain 模式（500ms），实现简单且规避 IPC 时序问题。
 */
export class VideoViewController {
  private view: WebContentsView | null = null

  /** 在主窗口打开视频页；已打开则先复用导航 */
  async open(win: BrowserWindow, url: string): Promise<void> {
    if (!this.view) {
      this.view = new WebContentsView({ webPreferences: { contextIsolation: true } })
      win.contentView.addChildView(this.view)
      this.view.setBounds({ x: 0, y: 48, width: win.getContentBounds().width, height: win.getContentBounds().height - 48 })
    }
    await this.view.webContents.loadURL(url)
    // 自动播放无需用户手势（成员端可能被动播放）
    await this.view.webContents.executeJavaScript('document.querySelector("video")?.play().catch(()=>{}); "ok"').catch(() => {})
  }

  /** 注入监听脚本；injectGuard 为真时追加跟随模式（成员端） */
  async inject(injectGuard = false): Promise<string> {
    if (!this.view) return 'noview'
    const r = await this.view.webContents.executeJavaScript(MONITOR_SCRIPT).catch(() => 'inject-error')
    if (injectGuard) await this.view.webContents.executeJavaScript(FOLLOWER_GUARD_SCRIPT).catch(() => {})
    return String(r)
  }

  /** 轮询并取走积压的视频事件 */
  async drainEvents(): Promise<Array<{ ev: string; position: number; paused: boolean }>> {
    if (!this.view) return []
    return this.view.webContents.executeJavaScript('window.__p2pBridge ? window.__p2pBridge.drain() : []').catch(() => [])
  }

  /** 查询视频当前状态（无视频返回 null） */
  async status(): Promise<{ position: number; paused: boolean; rate: number; duration: number } | null> {
    if (!this.view) return null
    return this.view.webContents.executeJavaScript('window.__p2pBridge ? window.__p2pBridge.status() : null').catch(() => null)
  }

  /** 向视频页下发指令 */
  async cmd(action: string, arg?: number): Promise<void> {
    if (!this.view) return
    await this.view.webContents.executeJavaScript(`window.__p2pBridge && window.__p2pBridge.cmd(${JSON.stringify(action)}, ${arg ?? 'null'})`).catch(() => {})
  }

  /** 视口随窗口尺寸调整 */
  resize(win: BrowserWindow): void {
    this.view?.setBounds({ x: 0, y: 48, width: win.getContentBounds().width, height: win.getContentBounds().height - 48 })
  }

  /** 释放视图 */
  close(): void {
    this.view?.webContents.close()
    this.view = null
  }
}
```

```ts
/**
 * 应用入口。
 * - --profile=<name>：切换 userData 目录，支持同机多实例联调（T1）
 * - p2psync:// 协议：Windows 下经 second-instance argv 传入，macOS 下经 open-url
 * - IPC 通道：openVideo / inject / drainEvents / videoStatus / videoCmd / copyText

// --profile 参数必须在 app.ready 前设置路径

if (profileArg) app.setPath('userData', app.getPath('userData') + '-' + profileArg.split('=')[1])

// 注册自定义协议（开发模式下需传 electron.exe 路径与参数，否则系统无法唤起）
if (app.isPackaged) app.setAsDefaultProtocolClient('p2psync')
else app.setAsDefaultProtocolClient('p2psync', process.execPath, [app.getAppPath()])

let mainWindow: BrowserWindow | null = null
const video = new VideoViewController()

/** 解析 argv 中的 p2psync:// 链接并推送给 UI */
function handleProtocolUrl(argv: string[]): void {
  const url = argv.find((a) => a.startsWith('p2psync://'))
  if (url && mainWindow) mainWindow.webContents.send('protocol-url', url)
}

/** 创建主窗口 */

  mainWindow = new BrowserWindow({
    width: 1280,
    height: 800,
    webPreferences: { preload: join('out/preload/index.js'), contextIsolation: true },

  mainWindow.on('resize', () => video.resize(mainWindow!))

  ipcMain.handle('openVideo', (_e, url: string) => mainWindow && video.open(mainWindow, url))
```

`src/room.ts`:

```ts
/**
 * UI 侧房间控制器：串联 shareLink/protocol/room/syncEngine 与 Electron API。
 * 职责：创建/加入房间、发送 hello、房主事件监听+广播、成员心跳校准循环、成员列表维护。
 */
import { computeTargetPosition, decideCorrection, type StateSnapshot } from '../core/syncEngine'
import { buildShareUrl, generateRoomId } from '../core/shareLink'
import type { SyncMsg } from '../core/protocol'
import { createRoom, openRealRoom, type TrysteroRoomLike } from '../core/room'

/** 房间角色 */
export type Role = 'host' | 'follower'

/** 成员展示信息 */
export interface MemberInfo {
  id: string
  role: Role | 'peer'
}

/** 房间控制器：UI 与 P2P/视频层之间的唯一中介 */
export class RoomController {
  role: Role = 'host'
  roomId = ''
  members = new Map<string, string>() // peerId → 显示名
  private room: ReturnType<typeof createRoom> | null = null
  private lastSnapshot: StateSnapshot | null = null
  private lastUrl = ''
  private heartbeatTimer: number | null = null

  /**
   * 创建房间（房主）。
   * 参数：videoUrl 当前视频页地址。
   * 返回值：分享链接。
   */
  async host(videoUrl: string): Promise<string> {
    this.role = 'host'
    this.roomId = generateRoomId()
    await this.attach()
    this.startHeartbeat()
    return buildShareUrl(this.roomId, videoUrl)
  }

  /**
   * 加入房间（成员）。
   * 参数：roomId 房间 ID。
   */
  async join(roomId: string): Promise<void> {
    this.role = 'follower'
    this.roomId = roomId
    await this.attach()
    // 立即请求全量状态
    this.room?.broadcast({ t: 'hello' })
  }

  /** 建立 P2P 房间并注册消息处理 */
  private async attach(): Promise<void> {
    const raw: TrysteroRoomLike = await openRealRoom(this.roomId)
    this.room = createRoom(raw, '', (msg, peerId) => this.handleMsg(msg, peerId))
    this.room.onPeerJoin((id) => this.members.set(id, '成员 ' + (this.members.size + 1)))
    this.room.onPeerLeave((id) => this.members.delete(id))
  }

  /**
   * 处理收到的同步消息。
   * 参数：msg 同步消息；peerId 发送方。
   */
  private handleMsg(msg: SyncMsg, peerId: string): void {
    if (this.role === 'host') {
      // 房主：响应 hello，回全量状态
      if (msg.t === 'hello') this.sendState()
      return
    }
    // 成员：应用房主指令
    if (msg.t === 'state') this.applySnapshot({ position: msg.position, playing: msg.playing, at: msg.at }, msg.url)
    if (msg.t === 'play') window.p2pApi.videoCmd('play')
    if (msg.t === 'pause') window.p2pApi.videoCmd('pause')
    if (msg.t === 'seek') window.p2pApi.videoCmd('seek', msg.position)
  }

  /**
   * 成员端应用状态快照：导航（必要时）+ 计算校准决策。
   * 参数：s 快照；url 房主当前视频页。
   */
  private async applySnapshot(s: StateSnapshot, url: string): Promise<void> {
    if (url && url !== this.lastUrl) {
      this.lastUrl = url
      await window.p2pApi.openVideo(url)
      await window.p2pApi.inject(true) // 成员端注入 + 跟随模式
      await window.p2pApi.videoCmd('seek', s.position)
      if (s.playing) await window.p2pApi.videoCmd('play')
    }
    this.lastSnapshot = s
  }

  /** 房主：开启周期心跳（2s）与本地事件上报轮询 */
  private startHeartbeat(): void {
    if (this.heartbeatTimer) return
    this.heartbeatTimer = window.setInterval(async () => {
      this.sendState()
      // 消费房主本机视频事件，转换为同步消息广播
      for (const ev of await window.p2pApi.drainEvents()) {
        const at = Date.now()
        if (ev.ev === 'play') this.room?.broadcast({ t: 'play', position: ev.position, at })
        if (ev.ev === 'pause') this.room?.broadcast({ t: 'pause', position: ev.position })
        if (ev.ev === 'seek') {
          const st = await window.p2pApi.videoStatus()
          this.room?.broadcast({ t: 'seek', position: ev.position, playing: !st?.paused, at })
        }
      }
    }, 2000)
  }

  /** 房主：广播全量状态 */
  private sendState(): void {
    window.p2pApi.videoStatus().then((st) => {
      if (!st || !this.room) return
      this.room.broadcast({ t: 'state', url: this.lastUrl, position: st.position, playing: !st.paused, at: Date.now() })
    })
  }

  /** 成员：开启校准循环（每 2s 对表一次） */
  startFollowLoop(): void {
    window.setInterval(async () => {
      if (this.role !== 'follower' || !this.lastSnapshot) return
      const target = computeTargetPosition(this.lastSnapshot, Date.now())
      const st = await window.p2pApi.videoStatus()
      if (!st) return
      const c = decideCorrection(target, st.position)
      if (c.kind === 'seek') await window.p2pApi.videoCmd('seek', c.position)
      if (c.kind === 'rate') await window.p2pApi.videoCmd('rate', c.rate)
    }, 2000)
  }

  /** 离开房间并清理定时器 */
  leave(): void {
    if (this.heartbeatTimer) clearInterval(this.heartbeatTimer)
    this.room?.leave()
    this.room = null
  }
}
```

```vue
  if (!parsed) return
  isHost.value = false
  roomId.value = parsed.roomId
  await controller.join(parsed.roomId)
  controller.startFollowLoop()
  refreshMembers()
}

/** 打开视频页（房主未建房时单独打开） */
async function openVideo(): Promise<void> {
  await window.p2pApi.openVideo(videoUrl.value)
  if (roomId.value && isHost.value) await window.p2pApi.inject(false)
}

/** 复制邀请链接 */
async function copyLink(): Promise<void> {
  await window.p2pApi.copyText(`p2psync://join?room=${roomId.value}&url=${encodeURIComponent(videoUrl.value)}`)
}

/** 刷新成员显示 */
function refreshMembers(): void {
  setInterval(() => (membersText.value = [...controller.members.values()].join('、') || '(等待成员)'), 1000)
}

onMounted(() => {
  // 系统唤起：second-instance/open-url 传来的邀请链接
  window.p2pApi.onProtocolUrl(async (url) => {
    joinInput.value = url
    await onJoinLink()
  })
})
</script>
```
